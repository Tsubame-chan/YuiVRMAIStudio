cmake_minimum_required(VERSION 3.22)
project(YuiLlamaAdapter LANGUAGES C CXX)
set(LLAMA_SOURCE_DIR "" CACHE PATH "Pinned llama.cpp source directory")
if(NOT EXISTS "${LLAMA_SOURCE_DIR}/include/llama.h")
  message(FATAL_ERROR "Provide the reviewed llama.cpp source with -DLLAMA_SOURCE_DIR=...")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
file(GLOB_RECURSE YUI_JINJA_SOURCES ../../../unity/Assets/Plugins/iOS/YuiLlamaJinja/*.cpp)
add_library(YuiLlama SHARED ../../../unity/Assets/Plugins/iOS/YuiLlamaBridge.cpp ${YUI_JINJA_SOURCES})
target_include_directories(YuiLlama PRIVATE ../../../unity/Assets/Plugins/iOS/YuiLlamaJinja)
target_compile_features(YuiLlama PRIVATE cxx_std_17)
target_link_libraries(YuiLlama PRIVATE llama)
set_target_properties(YuiLlama PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
